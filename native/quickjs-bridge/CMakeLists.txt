cmake_minimum_required(VERSION 3.20)
project(susu_quickjs C)
# Static CRT: no VC++ redistributable is required on a clean Windows 11 system.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT DEFINED QUICKJS_SOURCE)
  message(FATAL_ERROR "QUICKJS_SOURCE must point to the pinned source tree")
endif()
set(QJS_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(QJS_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory("${QUICKJS_SOURCE}" quickjs)
add_library(susu_quickjs SHARED bridge.c)
target_include_directories(susu_quickjs PRIVATE "${QUICKJS_SOURCE}")
target_link_libraries(susu_quickjs PRIVATE qjs)
set_property(TARGET susu_quickjs PROPERTY C_STANDARD 11)
